n    i     k     e

row1 first and then row2. Thus output = nikegtmc

/*Check Neon Number
A number is neon if the sum of digits of its square is equal to the number.
Example: 9² = 81 → 8 + 1 = 9

Test Case:
input=9
output=true

input=10
output=false
*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        int square=n*n;
        int sum=0;
        while(square>0){
            int digit=square%10;
            sum+=digit;
            square/=10;
        }
        if(sum==n){
            System.out.print("true");
        }else{
            System.out.print("false");
        }
    }
}

/*Create an abstract class Shape with abstract methods getArea() and getPerimeter(). 
Create subclasses Circle and Rectangle that implement these methods.
input=5
4
5
output=Area of Circle=78.5
Circumference of Circle=31.4
Area of Rectangle=20
Perimeter of Rectangle=18
*/
import java.util.*;
abstract class shape{
    abstract double getArea();
    abstract double getPerimeter();
}
class Circle extends shape{
    double radius;
    Circle(double radius){
        this.radius=radius;
    }
    double getArea(){
        return Math.PI * radius*radius;
    }
    double getPerimeter(){
        return 2* Math.PI * radius;
    }
    
}
class Rectangle extends shape{
    double length;
    double width;
    Rectangle(double length,double width){
        this.length=length;
        this.width=width;
    }
    double getArea(){
        return length*width;
    }
    double getPerimeter(){
        return (2*length)+(2*width);
    }
}
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        double rad=sc.nextDouble();
        double length=sc.nextDouble();
        double width=sc.nextDouble();
        Circle c=new Circle(rad);
        System.out.printf("Area of Circle=%.1f\n",c.getArea());
        System.out.printf("Circumference of Circle=%.1f\n",c.getPerimeter());
        Rectangle r=new Rectangle(length,width);
        System.out.println("Area of Rectangle="+r.getArea());
        System.out.println("Perimeter of Rectangle="+r.getPerimeter());
        
        
        
    }

import java.util.*;
abstract class BankAccount{
    protected double balance;
    BankAccount(double initialBalance){
        this.balance = initialBalance;
    }

    abstract void deposit(double amount);
    abstract void withdraw(double amount);
    double getBalance(){
        return balance;
    }
}
class SavingsAccount extends BankAccount{
    SavingsAccount(double initialBalance){
        super(initialBalance);
    }
    void deposit(double amount){
        balance += amount;
    }
    void withdraw(double amount){
        if (balance - amount >= 1000){
            balance -= amount;
        }else{
            System.out.println("Minimum balance of Rs.1000 required.");
        }
    }
}
class CurrentAccount extends BankAccount{
    CurrentAccount(double initialBalance){
        super(initialBalance);
    }
    void deposit(double amount){
        balance += amount;
    }
    void withdraw(double amount){
        balance -= amount;
    }
}
public class Test{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        SavingsAccount sa = new SavingsAccount(sc.nextDouble());
        sa.withdraw(sc.nextDouble());
        sa.deposit(sc.nextDouble());

        CurrentAccount ca = new CurrentAccount(sc.nextDouble());
        ca.withdraw(sc.nextDouble());
        ca.deposit(sc.nextDouble());

        System.out.println("Savings Balance: " + sa.getBalance());
        System.out.println("Current Balance: " + ca.getBalance());

Design an abstract class Course with an abstract method deliverContent() & a defined method displayType() to display course title. 
Each course must also display its course type. 
Subclasses include RecordedCourse and LiveCourse, each with custom behavior.

Testcases:

Course Title: Nil

case=4
input=Nil
Nil
ouput=Course Title: Nil
There is no pre-recorded video content.
Course Title: Nil

*/
import java.util.*;
abstract class Course{
    protected String title;
    Course(String title){
        this.title=title;
    }
    void displayType(){
        System.out.println("Course Title: "+title);
    }
    abstract void deliverContent();
}
class RecordedCourse extends Course{
    RecordedCourse(String title){
        super(title);
    }
    void deliverContent(){
        if(title.equals("Nil")){
            System.out.println("There is no pre-recorded video content.");
        }else{
            System.out.println("Streaming pre-recorded video content.");
            
        }
    }
}
class LiveCourse extends Course{
    LiveCourse(String title){
        super(title);
    }
    void deliverContent(){
        if(title.equals("Nil")){
            System.out.println("There is no live session planned.");
        }else{
            System.out.println("Conducting live session with instructor.");
            
        }
    }
}
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        RecordedCourse rc=new RecordedCourse(sc.nextLine());
        rc.displayType();
        rc.deliverContent();
        LiveCourse lc=new LiveCourse(sc.nextLine());
        lc.displayType();
        lc.deliverContent();
        
    }
}
