/*
Mr Somanath is a math teacher. He is given a problem to his students.
The students are given a number N, and the student has to perform following step:
       - Add each digit of the number and add the result to N
   Repeat this procedure until N becomes a single digit number.
    
 Your task is to help the students to perform the above steps and 
   print the resultant single digit number N.
    
    
    Input Format:
    -------------
    An integer, number N.
    
   Output Format:
   --------------
   Print an integer result.
    
    
   Sample Input-1:
   ---------------
    95
    
   Sample Output-1:
   ----------------
   5
   
   Explanation: 
   ------------
   95 => 9 + 5 = 14
   14 => 1 + 4 = 5
   Answer is 5
   
   
   Sample Input-2:
   ---------------
   765
   
   Sample Output-2:
   ----------------
    9
*/
import java.util.Scanner;

public class Test 
{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        int N = sc.nextInt();
        while (N >= 10){
            int sum = 0;
            while (N > 0){
                sum += N % 10;
                N /= 10;      
            }
            N = sum;
        }

        System.out.println(N);
    }

}

/*
Implement Word Frequency Analyzer : Read a line of strings and print the top 3 most frequent words.

Key Concepts:
- HashMap<String, Integer> for frequency count
- PriorityQueue<Map.Entry> with custom comparator

Test Case:
input=Java is simple. Java is powerful. Java is everywhere.
output=java: 3
is: 3
simple: 1
*/
import java.util.*;
public class Test {
    public static void main(String[] args){
        

        Scanner sc=new Scanner(System.in);
        String input =sc.nextLine();
        input = input.replaceAll("[^a-zA-Z ]", "").toLowerCase();
        String[] words = input.split("\\s+");
        Map<String, Integer> freqMap = new HashMap<>();
        for (String word : words){
            freqMap.put(word, freqMap.getOrDefault(word, 0) + 1);
        }
        PriorityQueue<Map.Entry<String, Integer>> pq = new PriorityQueue<>(
            (a, b) -> b.getValue() - a.getValue()  // Descending by frequency
        );
        pq.addAll(freqMap.entrySet());
        System.out.println("Top 3 frequent words:");
        int count = 0;
        while (!pq.isEmpty() && count < 3){
            Map.Entry<String, Integer> entry = pq.poll();
            System.out.println(entry.getKey() + ": " + entry.getValue());
            count++;
        }
    }

}

        String input=sc.next();
        Map<Character,Integer> freq=new LinkedHashMap<>();
        for(char ch:input.toCharArray()){
            freq.put(ch,freq.getOrDefault(ch,0)+1);
            
        }
        char result='-';
        for(char ch:freq.keySet()){
            if(freq.get(ch)==1){
                result=ch;
                break;
            }
        }
        if(result=='-'){
            System.out.println("-1");
        }else{
            System.out.println("First non-repeating character:"+result);
        }
    }

/*Create an abstract class Shape with abstract methods getArea() and getPerimeter(). 
Create subclasses Circle and Rectangle that implement these methods.
input=5
4
5
output=Area of Circle=78.5
Circumference of Circle=31.4
Area of Rectangle=20
Perimeter of Rectangle=18
*/
import java.util.*;
abstract class shape{
    abstract double getArea();
    abstract double getPerimeter();
}
class Circle extends shape{
    double radius;
    Circle(double radius){
        this.radius=radius;
    }
    double getArea(){
        return Math.PI * radius*radius;
    }
    double getPerimeter(){
        return 2* Math.PI * radius;
    }
    
}
class Rectangle extends shape{
    double length;
    double width;
    Rectangle(double length,double width){
        this.length=length;
        this.width=width;
    }
    double getArea(){
        return length*width;
    }
    double getPerimeter(){
        return (2*length)+(2*width);
    }
}
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        double rad=sc.nextDouble();
        double length=sc.nextDouble();
        double width=sc.nextDouble();
        Circle c=new Circle(rad);
        System.out.printf("Area of Circle=%.1f\n",c.getArea());
        System.out.printf("Circumference of Circle=%.1f\n",c.getPerimeter());
        Rectangle r=new Rectangle(length,width);
        System.out.println("Area of Rectangle="+r.getArea());
        System.out.println("Perimeter of Rectangle="+r.getPerimeter());
        
        
        
    }
}

    int prevread;
    int curread;
    double totalUnits;
    double totalBill;

        this.consnum = consnum;
        this.type = Character.toUpperCase(type);
        this.name = name;
        this.prevread = prevread;
        this.curread = curread;
        this.totalUnits = curread - prevread;

            else if (totalUnits <= 200)
                totalBill = totalUnits * 4;
            else if (totalUnits <= 500)
                totalBill = totalUnits * 6;
            else
                totalBill = totalUnits * 8;
        }else{
            System.out.println("Invalid connection type.");
        }
    }

        System.out.println("\n\tConsumer number = " + consnum);
        System.out.println("\tConsumer name = " + name);

public class Test {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Input array size
        int n = sc.nextInt();
        int[] arr = new int[n];

        // Input array elements
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        // Input the digit k
        int k = sc.nextInt();

        // Print elements containing k as one of the digits
        for (int i = 0; i < n; i++) {
            if (containsDigit(arr[i], k)) {
                System.out.print(arr[i] + " ");
            }
        }

        sc.close();
    }

    // Method to check if a number contains the digit k
    public static boolean containsDigit(int num, int k) {
        while (num > 0) {
            if (num % 10 == k) {
                return true;
            }
            num /= 10;
        }
        return false;
    }
}

/* WAJP to check whether the 2 given numbers are with same digit 
ot not. If yes print true else false.
Example :   14 & 54 - common digit is 4 - output : TRUE
            23 & 45 - No common digit - output : FALSE
*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int n1=sc.nextInt();
        System.out.print(" ");
        int n2=sc.nextInt();
        String str1=String.valueOf(n1);
        String str2=String.valueOf(n2);
        boolean found=false;
        for(int i=0;i<str1.length();i++){
            char ch=str1.charAt(i);
            if(str2.indexOf(ch)!=-1){
                found=true;
                break;
            }
        }
        System.out.print(found?"TRUE":"FALSE");
    }
}

/*
Two friends are playing a game in which they will say any name. 
If that name contains all unique characters then he is the winner, 
otherwise looser.

Example 1
Enter name:
Ngit                                                                                                                
You won the game!!

Example 2
Enter name:
Praveen                                                                                                                 
You lost the game
*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        System.out.println("Enter name:");
        String name=sc.next();
        name=name.toLowerCase();
        boolean unique=true;
        Set<Character> set=new HashSet<>();
        
        for(int i=0;i<name.length();i++){
            char ch=name.charAt(i);
            if(set.contains(ch)){
                unique=false;
                break;
            }
            set.add(ch);
        }
        if(unique){
            System.out.println("You won the game!!");
        }
        else{
            System.out.println("You lost the game");
            
        }
        
        
        
        
    }
}

/*
WAJP to accept consumer number,Type of connection(D-Domestic/C-Commercial),consumer name,previous month reading & current month reading from the user. Then calculate electricity bill by deducting previous reading from current reading. Now print the total bill with following constraints.
if the connection is Domestic
	total units<=100 multiply with 1
	total units>100 && units<=200 multiply with 3
	total units>200 && units<=500 multiply with 5
	total units>500 multiply with 7
if the connection is Commercial
	total units<=100 multiply with 2
	total units>100 && units<=200 multiply with 4
	total units>200 && units<=500 multiply with 6
	total units>500 multiply with 8
input=
Enter consumer number  456
Enter Type(D or C) of connection  C
Enter consumer name  RadhaKrishna
Enter previous month reading  125
Enter current month reading 525
output=         
	 Consumer number = 456
         Consumer name = RadhaKrishna
         Type of connection  = COMMERCIAL
         Previous Month  Reading = 125.0
         Current Month  Reading = 525.0
         Total units = 400.0
         Total bill = RS 2400.0

case=2
input=
Enter consumer number  12345
Enter Type(D or C) of connection  D
Enter consumer name  RamaKrishna
Enter previous month reading  250
Enter current month reading 350

output=
         Consumer number = 12345
         Consumer name = RamaKrishna
         Type of connection  = DOMESTIC
         Previous Month  Reading = 250.0
         Current Month  Reading = 350.0
         Total units = 100.0
         Total bill = RS 100.0
*/
import java.util.*;
class Bill{
    int consnum;
    char type;
    String name;
    int prevread;
    int curread;
    double totalUnits;
    double totalBill;
    Bill(int consnum, char type, String name, int prevread, int curread){
        this.consnum = consnum;
        this.type = Character.toUpperCase(type);
        this.name = name;
        this.prevread = prevread;
        this.curread = curread;
        this.totalUnits = curread - prevread;
        calculateBill();
    }
    private void calculateBill(){
        if (type == 'D'){
            if (totalUnits <= 100)
                totalBill = totalUnits * 1;
            else if (totalUnits <= 200)
                totalBill = totalUnits * 3;
            else if (totalUnits <= 500)
                totalBill = totalUnits * 5;
            else
                totalBill = totalUnits * 7;
        }else if(type == 'C'){
            if (totalUnits <= 100)
                totalBill = totalUnits * 2;
            else if (totalUnits <= 200)
                totalBill = totalUnits * 4;
            else if (totalUnits <= 500)
                totalBill = totalUnits * 6;
            else
                totalBill = totalUnits * 8;
        }else{
            System.out.println("Invalid connection type.");
        }
    }
    public void printDetails(){
        System.out.println("\n\tConsumer number = " + consnum);
        System.out.println("\tConsumer name = " + name);
        System.out.println("\tType of connection  = " + (type == 'D' ? "DOMESTIC" : "COMMERCIAL"));
        System.out.println("\tPrevious Month  Reading = " + (double) prevread);
        System.out.println("\tCurrent Month  Reading = " + (double) curread);
        System.out.println("\tTotal units = " + totalUnits);
        System.out.println("\tTotal bill = RS " + totalBill);
    }
}

        System.out.print("Enter Type(D or C) of connection  ");

        System.out.print("Enter previous month reading  ");
        int prevread = sc.nextInt();

        System.out.print("Enter current month reading  ");
        int curread = sc.nextInt();

        b.printDetails();
