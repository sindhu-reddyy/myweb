/*Problem: Sort a list of strings in alphabetical order.
input=radha krishna madhu sudha asshu dhudhu
output=Sorted Names: asshu dhudhu krishna madhu radha sudha
*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        String input=sc.nextLine();
        String names[]=input.split(" ");
        Arrays.sort(names);
        System.out.print("Sorted Names: ");

            System.out.print(name+" ");
        }
    }

/*Problem: Find First Non-Repeating Character
Testcase:
input=jakejakeradha
output=First non-repeating character: r

input=nnggiitt
output=-1
*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        String input=sc.next();
        Map<Character,Integer> freq=new LinkedHashMap<>();
        for(char ch:input.toCharArray()){
            freq.put(ch,freq.getOrDefault(ch,0)+1);
            
        }
        char result='-'; 
        for(char ch:freq.keySet()){
            if(freq.get(ch)==1){
                result=ch;
                break;
            }
        }
        if(result=='-'){
            System.out.println("-1");
        }else{
            System.out.println("First non-repeating character:"+result);
        }
    }
}

/*Problem: Given a list of words, group them based on their lengths.
Testcase:
input=java is fun to code and happy to execute
output=Grouped by Length: {2=[is, to, to], 3=[fun, and], 4=[java, code], 5=[happy], 7=[execute]}

input=all is well and all the best
output=Grouped by Length: {2=[is], 3=[all, and, all, the], 4=[well, best]}

*/
import java.util.*;
class Test{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        String input=sc.nextLine();
        String[] words=input.split(" ");
        Map<Integer,List<String>> grouped=new TreeMap<>();
        for(String word:words){
            int len=word.length();
            grouped.putIfAbsent(len,new ArrayList<>());
            grouped.get(len).add(word);
        }
        System.out.println("Grouped by Length:"+grouped);
    }
}

/*
Implement stack operations using List interface
Testcase:
input=10 20 30 40 50
output=Stack Elements = 50 40 30 20 10
Stack Size = 5
Peek Element = 50
Popped Top = 50
Popped Top = 40
Stack Elements = 30 20 10
Stack Size = 3
*/
import java.util.*;
public class Test{
    public static void main(String[] args){
        
        Stack<Integer> stack=new Stack<>();
        Scanner sc=new Scanner(System.in);
        String[] input=sc.nextLine().split(" ");
        
        
        for(String str:input){
            stack.add(Integer.parseInt(str));
        }
        System.out.print("Stack Elements = ");
        for(int i=stack.size()-1;i>=0;i--)System.out.print(stack.get(i)+" ");
        System.out.println("\nStack Size = "+stack.size());
        System.out.println("Peek Element = "+stack.peek());
        System.out.println("Popped Top = "+stack.pop());
        System.out.println("Popped Top = "+stack.pop());
        
        System.out.print("Stack Elements = ");
        for(int i=stack.size()-1;i>=0;i--) System.out.print(stack.get(i)+" ");
        System.out.println("\nStack Size = "+stack.size());
        sc.close();
        
    }
}
