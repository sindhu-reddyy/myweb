import java.util.*;
class Solution
{
    static int numIslands(int[][] grid) {
        int n=grid.length;
        int m=grid[0].length;
        boolean[][] visited=new boolean[n][m];
        int count=0;
        int[] dr={-1,1,0,0};
        int[] dc={0,0,-1,1};
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(grid[i][j]==1 && !visited[i][j]){
                    count++;
                    Stack<int[]> stack=new Stack<>();
                    stack.push(new int[]{i,j});
                    visited[i][j]=true;
                    while(!stack.isEmpty()){
                        int[] cell=stack.pop();
                        int r=cell[0],c=cell[1];
                        for(int k=0;k<4;k++){
                            int nr=r+dr[k];
                            int nc=c+dc[k];
                            if(nr>=0 && nr<n && nc>=0 && nc<m && grid[nr][nc]==1 && !visited[nr][nc]){
                                visited[nr][nc]=true;
                                stack.push(new int[]{nr,nc});
                            }
                        }
                    }
                }
            }
        }
        return count;
    }
}


class Solution
{
     static int maxAreaOfIsland(int[][] grid) {
         int n=grid.length;
         int m=grid[0].length;
         int maxarea=0;
         for(int i=0;i<n;i++){
             for(int j=0;j<m;j++){
                 if(grid[i][j]==1){
                     int area=dfs(grid,i,j);
                     maxarea=Math.max(maxarea,area);
                 }
             }
         }
         return maxarea;
         
     }
     static int dfs(int[][] grid,int i,int j){
         if(i<0||i>=grid.length||j<0||j>=grid[0].length||grid[i][j]==0) return 0;
         grid[i][j]=0;
         int area=1;
         area+=dfs(grid,i-1,j);
         area+=dfs(grid,i+1,j);
         area+=dfs(grid,i,j-1);
         area+=dfs(grid,i,j+1);
         return area;
     }
}
